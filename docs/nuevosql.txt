-- ATENCIÓN: Esto borrará los datos actuales para asegurar que no haya errores de caché.

-- 1. Limpieza (Borrar para recrear correctamente)
DROP TABLE IF EXISTS playlist_songs;
DROP TABLE IF EXISTS playlists;
DROP TABLE IF EXISTS songs;
DROP TABLE IF EXISTS users;

-- 2. Crear Tabla de Usuarios
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'User',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Crear Tabla de Canciones (Asegurando la columna user_id)
CREATE TABLE songs (
    id BIGSERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    artist TEXT NOT NULL,
    audio_url TEXT NOT NULL,
    cover_url TEXT,
    duration TEXT,
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. Crear Tabla de Playlists
CREATE TABLE playlists (
    id BIGSERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. Crear Tabla Intermedia de Playlists
CREATE TABLE playlist_songs (
    id BIGSERIAL PRIMARY KEY,
    playlist_id BIGINT REFERENCES playlists(id) ON DELETE CASCADE,
    song_id BIGINT REFERENCES songs(id) ON DELETE CASCADE,
    position INTEGER NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(playlist_id, song_id)
);

-- 6. Habilitar Seguridad (RLS) en todas las tablas
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE songs ENABLE ROW LEVEL SECURITY;
ALTER TABLE playlists ENABLE ROW LEVEL SECURITY;
ALTER TABLE playlist_songs ENABLE ROW LEVEL SECURITY;

-- 7. Crear Políticas de Acceso Público (Para simplificar tu desarrollo)

-- Políticas para 'users'
CREATE POLICY "Public Insert Users" ON users FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Select Users" ON users FOR SELECT USING (true);
CREATE POLICY "Public Delete Users" ON users FOR DELETE USING (true);

-- Políticas para 'songs'
CREATE POLICY "Public Insert Songs" ON songs FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Select Songs" ON songs FOR SELECT USING (true);
CREATE POLICY "Public Delete Songs" ON songs FOR DELETE USING (true);

-- Políticas para 'playlists'
CREATE POLICY "Public All Playlists" ON playlists FOR ALL USING (true) WITH CHECK (true);

-- Políticas para 'playlist_songs'
CREATE POLICY "Public All Playlist Songs" ON playlist_songs FOR ALL USING (true) WITH CHECK (true);
